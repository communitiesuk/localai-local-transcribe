Guardrail Process Flow

Traces the flow from PostgreSQL table definitions through backend processing to frontend GUI display.

1. Database Layer (PostgreSQL)

Table Definition: guardrail_result

Location: common/database/postgres_models.py
class GuardrailResult(BaseTableMixin, table=True):
    __tablename__ = "guardrail_result"
    created_datetime: datetime
    updated_datetime: datetime
    minute_version_id: UUID | None  # Foreign key to minute_version
    minute_version: "MinuteVersion"  # Relationship back to MinuteVersion
    passed: bool                      # Whether the check passed
    score: float | None               # Confidence score (0.0 - 1.0)
    reasoning: str | None             # Explanation for the result
    error: str | None                 # Error message if check failed

2. Backend Processing Layer

Step 1: Minute Generation Trigger

Location: common/services/minute_handler_service.py
async def process_minute_generation_message(cls, minute_version_id: UUID) -> None:
    # 1. Generate the minute content
    result = await cls.generate_minutes(meeting_type, minute_version.minute)
    html_content = result.text
    
    # 2. Run guardrail check (if transcript is long enough)
    await cls._run_accuracy_guardrail(
        minute_version_id=minute_version.id,
        minute_id=minute_version.minute_id,
        content=html_content,
        transcript=dialogue_entries,
        label="generation",
    )
! Warning The guardrail check is skipped if the transcript word count is below 200 words (MIN_WORD_COUNT_FOR_FULL_SUMMARY).

Step 2: Accuracy Calculation

async def calculate_accuracy_score(cls, minute, transcript) -> GuardrailScore:
    chatbot = create_default_chatbot(FastOrBestLLM.FAST)
    return await chatbot.structured_chat(
        messages=get_accuracy_check_messages(minute, transcript),
        response_format=GuardrailScore,
    )

3. API Layer (FastAPI)

Endpoint: List Minute Versions

Location: backend/api/routes/minutes.py
@minutes_router.get("/minutes/{minute_id}/versions")
async def list_minute_versions(...) -> list[MinuteVersionResponse]:
    # Calculate if meeting is too short for summary
    word_count = sum(len(entry["text"].split()) for entry in dialogue_entries)
    is_too_short = word_count < settings.MIN_WORD_COUNT_FOR_SUMMARY

    return [
        MinuteVersionResponse(
            ...,
            too_short=is_too_short,
            guardrail_results=[...]
        )
    ]
